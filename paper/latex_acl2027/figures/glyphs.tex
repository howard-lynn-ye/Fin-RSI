% Glyphs for the method figures (episode, gate, arms). Pictures first, words only as labels.%
\definecolor{gink}{HTML}{1F2335}%
\definecolor{gmid}{HTML}{6A7385}%
\definecolor{gline}{HTML}{C9D0DA}%
\definecolor{gok}{HTML}{1E9E6A}%
\definecolor{gbad}{HTML}{D2483E}%
\providecommand{\glab}{\fontsize{7.5}{9}\selectfont}%
\providecommand{\gsmall}{\fontsize{7}{8.4}\selectfont}%
\tikzset{%
  gword/.style={font=\sffamily\glab, text=gink, inner sep=0, align=center},%
  gnote/.style={font=\sffamily\gsmall, text=gmid, inner sep=0, align=center},%
  gflow/.style={-{Latex[length=1.7mm,width=1.4mm]}, line width=0.75pt, draw=gmid!70},%
  gshade/.style={blur shadow={shadow blur steps=6, shadow xshift=0pt, shadow yshift=-0.7pt,%
                 shadow opacity=18, shadow blur radius=1.8pt}},%
}%
% \gpage{centre}{w}{h}{fill}: a page with a folded corner.%
\providecommand{\gpage}[4]{%
  \begin{scope}[shift={#1}]%
    \path[fill=#4, draw=gline, line width=0.5pt, line join=round, gshade]%
      ({-#2/2},{-#3/2}) -- ({#2/2},{-#3/2}) -- ({#2/2},{#3/2-0.2}) -- ({#2/2-0.2},{#3/2}) -- ({-#2/2},{#3/2}) -- cycle;%
    \draw[gline, line width=0.5pt, line join=round, fill=gline!40] ({#2/2-0.2},{#3/2}) -- ({#2/2-0.2},{#3/2-0.2}) -- ({#2/2},{#3/2-0.2});%
  \end{scope}}%
% \gbar{x0}{y}{length}{colour}: one line of text or code, drawn as a rounded bar.%
\providecommand{\gbar}[4]{\draw[#4, line width=2.4pt, line cap=round] (#1,#2) -- ({#1+#3},#2);}%
% \gcheck{centre}{state}: a check that passed (1) or failed (0).%
\providecommand{\gcheck}[2]{%
  \ifnum#2=1 \fill[gok] #1 circle (0.13);%
    \draw[white, line width=0.8pt, line cap=round, line join=round] ($#1+(-0.06,0)$) -- ($#1+(-0.015,-0.05)$) -- ($#1+(0.065,0.055)$);%
  \else \fill[gbad] #1 circle (0.13);%
    \draw[white, line width=0.8pt, line cap=round] ($#1+(-0.05,-0.05)$) -- ($#1+(0.05,0.05)$) ($#1+(-0.05,0.05)$) -- ($#1+(0.05,-0.05)$);\fi}%
% \gcard{centre}: a rule card, small and blue.%
\providecommand{\gcard}[1]{%
  \begin{scope}[shift={#1}]%
    \path[fill=finblue!12, draw=finblue!70, line width=0.55pt, rounded corners=1pt, gshade] (-0.22,-0.16) rectangle (0.22,0.16);%
    \draw[finblue!70, line width=0.9pt, line cap=round] (-0.13,0.05) -- (0.13,0.05) (-0.13,-0.05) -- (0.06,-0.05);%
  \end{scope}}%
% \glock{centre}: a padlock.%
\providecommand{\glock}[1]{%
  \begin{scope}[shift={#1}]%
    \draw[gmid, line width=0.9pt] (-0.09,0.05) -- (-0.09,0.13) arc[start angle=180, end angle=0, radius=0.09] -- (0.09,0.05);%
    \fill[gmid, rounded corners=1pt] (-0.15,-0.15) rectangle (0.15,0.07);%
    \fill[white] (0,-0.03) circle (0.028);%
  \end{scope}}%
% \gbot{centre}{radius}: the model, a dark disc with the robot icon.%
\providecommand{\gbot}[2]{%
  \path[fill=gink, gshade] #1 circle (#2);%
  \pic[draw=white, fill=white, line width=0.6pt, scale=1.25] at #1 {fin robot};}%
