% \fstrip{x0}{x1}{ytop}{text}: the grey title strip of a card.
\providecommand{\fstrip}[4]{%
  \begin{scope}%
    \clip[rounded corners=1.5pt] (#1,{#3-0.32}) rectangle (#2,#3);%
    \fill[frule!55] (#1,{#3-0.32}) rectangle (#2,#3);%
  \end{scope}%
  \draw[frule, line width=0.4pt] (#1,{#3-0.32}) -- (#2,{#3-0.32});%
  \node[anchor=west, font=\fcode, text=fmid, inner sep=0] at ({#1+0.1},{#3-0.16}) {#4};}%
