% \plane{z}{accent}: a thin sheet, its top face, front and right edges.
\newcommand{\plane}[2]{%
  \fill[#2!38] (0,0,{#1-0.06}) -- (\L,0,{#1-0.06}) -- (\L,0,#1) -- (0,0,#1) -- cycle;
  \fill[#2!48] (\L,0,{#1-0.06}) -- (\L,\D,{#1-0.06}) -- (\L,\D,#1) -- (\L,0,#1) -- cycle;
  \shade[left color=#2!7, right color=#2!14, opacity=0.95]
    (0,0,#1) -- (\L,0,#1) -- (\L,\D,#1) -- (0,\D,#1) -- cycle;
  \draw[#2!45, line width=0.35pt, line join=round]
    (0,0,{#1-0.06}) -- (\L,0,{#1-0.06}) -- (\L,\D,{#1-0.06}) -- (\L,\D,#1) -- (0,\D,#1)
    -- (0,0,#1) -- cycle (0,0,#1) -- (\L,0,#1) -- (\L,\D,#1);}
% \block{x}{y}{z}{w}{d}{h}{accent}{front fill}{front content}: a box standing on a plane,
% with a soft shadow cast back and to the right. Content is drawn in front-face coordinates.
\newcommand{\block}[9]{%
  \fill[black, opacity=0.07] ({#1+0.06},{#2+0.08},#3) -- ({#1+#4+0.18},{#2+0.08},#3)
    -- ({#1+#4+0.18},{#2+#5+0.3},#3) -- ({#1+0.06},{#2+#5+0.3},#3) -- cycle;
  \filldraw[fill=#7!14, draw=#7!55, line width=0.35pt, line join=round]
    (#1,#2,{#3+#6}) -- ({#1+#4},#2,{#3+#6}) -- ({#1+#4},{#2+#5},{#3+#6}) -- (#1,{#2+#5},{#3+#6}) -- cycle;
  \filldraw[fill=#7!30, draw=#7!55, line width=0.35pt, line join=round]
    ({#1+#4},#2,#3) -- ({#1+#4},{#2+#5},#3) -- ({#1+#4},{#2+#5},{#3+#6}) -- ({#1+#4},#2,{#3+#6}) -- cycle;
  \filldraw[fill=#8, draw=#7!55, line width=0.35pt, line join=round]
    (#1,#2,#3) -- ({#1+#4},#2,#3) -- ({#1+#4},#2,{#3+#6}) -- (#1,#2,{#3+#6}) -- cycle;
  \begin{scope}[canvas is xz plane at y=#2, shift={(#1,#3)}] #9 \end{scope}}
% \unit{x}{z}{w}{h}{icon}{label}{accent}: an icon and a label on a front face.
\newcommand{\facelabel}[5]{%
  \pic[draw=#5!80!black, fill=#5!80!black, line width=0.55pt, scale=0.9] at (0.32, #2/2) {#3};
  \node[anchor=west, font=\scriptsize\bfseries, text=ink, inner sep=0pt] at (0.58, #2/2) {#4};}
