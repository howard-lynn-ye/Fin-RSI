% \fslab{origin}{z}{colour}{label}: a W x D x T slab with its label printed on the front face.
\def\sW{3.45}\def\sD{1.85}\def\sT{0.36}
\newcommand{\fslab}[4]{%
  \begin{scope}[shift={#1}, iso]
    \coordinate (a) at (0,0,#2);
    \coordinate (b) at (\sW,0,#2);
    \coordinate (c) at (\sW,\sD,#2);
    \coordinate (at) at (0,0,{#2+\sT});
    \coordinate (bt) at (\sW,0,{#2+\sT});
    \coordinate (ct) at (\sW,\sD,{#2+\sT});
    \coordinate (dt) at (0,\sD,{#2+\sT});
  \end{scope}
  \fill[#3!78!black] (b) -- (c) -- (ct) -- (bt) -- cycle;          % right face
  \fill[#3] (a) -- (b) -- (bt) -- (at) -- cycle;                   % front face
  \fill[#3!16] (at) -- (bt) -- (ct) -- (dt) -- cycle;              % top face
  \draw[#3!70!black, line width=0.45pt, line join=round]
    (a) -- (b) -- (c) -- (ct) -- (dt) -- (at) -- cycle (at) -- (bt) -- (ct) (bt) -- (b);
  \begin{scope}[shift={(a)}, cm={0.95,-0.24,0,1,(0,0)}]
    \node[transform shape, text=white, font=\scriptsize\bfseries, anchor=center]
      at (\sW/2, \sT/2) {#4};
  \end{scope}}
% \token{point}{colour}{icon}: a pictogram standing on a top face.
\newcommand{\token}[3]{%
  \node[circle, fill=white, draw=#2!70!black, line width=0.45pt, minimum size=0.46cm,
        inner sep=0pt, copy shadow={shadow xshift=0.6pt, shadow yshift=-0.9pt, fill=#2!45,
        draw=none}] at #1 {};
  \pic[draw=#2!80!black, fill=#2!80!black, line width=0.55pt] at #1 {#3};}
