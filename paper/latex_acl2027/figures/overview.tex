\newcommand{\fincard}[7][4.7cm]{%  [text width] name, west corner, colour, icon, title, lines
  \node[cardtext=#1, anchor=north west] (#2) at ($#3+(0.66,0)$) {\textbf{#6}\\#7};
  \coordinate (#2L) at #3;
  \begin{scope}[on background layer]
    \path[draw=fingray!35, fill=white, rounded corners=3pt,
          drop shadow={shadow xshift=0.6pt, shadow yshift=-0.8pt, opacity=0.16}]
      (#2L |- #2.south) rectangle (#2.north east);
    \begin{scope}
      \clip[rounded corners=3pt] (#2L |- #2.south) rectangle (#2.north east);
      \fill[#4] (#2L |- #2.south) rectangle ($(#2L |- #2.north) + (0.58, 0)$);
    \end{scope}
  \end{scope}
  \pic[draw=white, fill=white, line width=0.55pt] at ($(#2L |- #2.center) + (0.29, 0)$) {#5};
}
\newcommand{\finbox}[5]{%  name, centre, icon, icon colour, text
  \node[box] (#1) at #2 {\hspace*{0.36cm}\begin{tabular}{@{}l@{}}#5\end{tabular}};
  \pic[draw=#4, fill=#4, line width=0.55pt] at ($(#1.west) + (0.27, 0)$) {#3};
}
