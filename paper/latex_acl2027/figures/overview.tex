% Left, the interfaces that bring information in; right, the interfaces the agent is used
% through. Counts are macros from numbers.tex. Oblique projection: x across, y into the page,
% z up, so front faces and their text are not distorted. Needs the TikZ libraries 3d,
% shadows.blur and fadings.

% ================================================================== helpers
% \card{x}{y centre}{w}{h}{accent}{icon}{label}: a white card with a tinted icon disc.
\newcommand{\card}[7]{%
  \path[card] (#1,{#2-#4/2}) rectangle ({#1+#3},{#2+#4/2});
  \fill[#5!14] ({#1+0.36},#2) circle (0.21);
  \pic[draw=#5, fill=#5, line width=0.55pt, scale=0.85] at ({#1+0.36},#2) {#6};
  \node[anchor=west, font=\sffamily\scriptsize\bfseries, text=ink, inner sep=0pt] at ({#1+0.68},#2) {#7};}

% \plate{z}{accent}{label}: a glass plate with its name on the front edge.
\newcommand{\plate}[3]{%
  \fill[#2!26] (0,0,#1) -- (\L,0,#1) -- (\L,0,{#1+\T}) -- (0,0,{#1+\T}) -- cycle;
  \fill[#2!40] (\L,0,#1) -- (\L,\D,#1) -- (\L,\D,{#1+\T}) -- (\L,0,{#1+\T}) -- cycle;
  \fill[#2!7, opacity=0.92] (0,0,{#1+\T}) -- (\L,0,{#1+\T}) -- (\L,\D,{#1+\T}) -- (0,\D,{#1+\T}) -- cycle;
  \draw[#2!50, line width=0.35pt, line join=round]
    (0,0,#1) -- (\L,0,#1) -- (\L,\D,#1) -- (\L,\D,{#1+\T}) -- (0,\D,{#1+\T}) -- (0,0,{#1+\T}) -- cycle
    (0,0,{#1+\T}) -- (\L,0,{#1+\T}) -- (\L,\D,{#1+\T}) (\L,0,{#1+\T}) -- (\L,0,#1);
  \draw[white, opacity=0.8, line width=0.5pt] (0.04,0,{#1+\T-0.02}) -- (\L-0.04,0,{#1+\T-0.02});
  \begin{scope}[canvas is xz plane at y=0]
    \node[font=\sffamily\tiny\bfseries, text=#2!72!black, inner sep=0pt] at (\L/2, {#1+\T/2}) {#3};
  \end{scope}
  % the core passes through the plate
  \begin{scope}[canvas is xy plane at z={#1+\T}]
    \fill[ink!18] (\cx,\cy) circle (0.075); \fill[ink!80] (\cx,\cy) circle (0.035);
  \end{scope}}
% \core{z from}{z to}: one visible segment of the core.
\newcommand{\core}[2]{%
  \draw[ink!85, line width=1.1pt] (\cx,\cy,#1) -- (\cx,\cy,#2);
  \draw[white, opacity=0.55, line width=0.3pt] ({\cx-0.012},\cy,#1) -- ({\cx-0.012},\cy,#2);}
% \wire{x}{z}{accent}: a chosen piece wired to the core along its plate.
\newcommand{\wire}[3]{%
  \begin{scope}[canvas is xy plane at z=#2]
    \draw[#3, line width=0.6pt, line cap=round] (#1,\ty) -- (\cx,\cy);
  \end{scope}}
% \cube{x}{y}{z}{w}{d}{h}{edge}{front}{top}{side}: a small box.
\newcommand{\cube}[9]{%
  \filldraw[fill=#9, draw=#7, line width=0.35pt, line join=round]
    (#1,#2,{#3+#6}) -- ({#1+#4},#2,{#3+#6}) -- ({#1+#4},{#2+#5},{#3+#6}) -- (#1,{#2+#5},{#3+#6}) -- cycle;
  \filldraw[fill=#8!80!black, draw=#7, line width=0.35pt, line join=round]
    ({#1+#4},#2,#3) -- ({#1+#4},{#2+#5},#3) -- ({#1+#4},{#2+#5},{#3+#6}) -- ({#1+#4},#2,{#3+#6}) -- cycle;
  \filldraw[fill=#8, draw=#7, line width=0.35pt, line join=round]
    (#1,#2,#3) -- ({#1+#4},#2,#3) -- ({#1+#4},#2,{#3+#6}) -- (#1,#2,{#3+#6}) -- cycle;}
